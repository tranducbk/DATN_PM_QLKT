\noindent\textbf{Thuật ngữ kỹ thuật}
\vspace{0.2cm}

\begin{longtable}{|>{\centering\arraybackslash}p{3.6cm}|R{10.5cm}|}
\hline
\textbf{Thuật ngữ} & \multicolumn{1}{c|}{\textbf{Ý nghĩa}} \\ \hline
\endfirsthead
\hline
\textbf{Thuật ngữ} & \multicolumn{1}{c|}{\textbf{Ý nghĩa}} \\ \hline
\endhead
\hline
\endfoot
\textbf{Strategy}              & Mẫu thiết kế hành vi cho phép định nghĩa một họ thuật toán có thể thay thế lẫn nhau, mỗi loại nghiệp vụ cài đặt một biến thể riêng. \\ \hline
\textbf{Registry}              & Bảng đăng ký trung tâm ánh xạ định danh sang đối tượng xử lý tương ứng (vd: ánh xạ loại đề xuất sang Strategy xử lý). \\ \hline
\textbf{Repository}            & Lớp trừu tượng đứng giữa tầng nghiệp vụ và cơ sở dữ liệu, đóng gói các thao tác đọc/ghi để tầng nghiệp vụ không phụ thuộc trực tiếp vào ORM. \\ \hline
\textbf{Middleware}            & Hàm trung gian trong chuỗi xử lý yêu cầu HTTP của Express, có thể xác thực, kiểm tra dữ liệu, ghi nhật ký hoặc chuyển tiếp sang hàm tiếp theo. \\ \hline
\textbf{Server Components}     & Thành phần React mặc định chạy phía máy chủ trong mô hình Next.js App Router, có thể truy vấn cơ sở dữ liệu trực tiếp trong hàm render. \\ \hline
\textbf{Migration}             & Tệp SQL được sinh tự động khi thay đổi lược đồ cơ sở dữ liệu, dùng để đồng bộ giữa các môi trường và truy vết lịch sử thay đổi schema. \\ \hline
\textbf{Refresh rotation}      & Cơ chế phát hành đồng thời cặp Access + Refresh Token mới mỗi lần làm mới phiên đăng nhập, đồng thời vô hiệu hoá Refresh Token cũ để hạn chế nguy cơ token bị đánh cắp. \\ \hline
\textbf{Transaction}           & Đơn vị giao dịch CSDL gồm một hoặc nhiều thao tác phải được thực thi trọn vẹn (hoặc tất cả thành công, hoặc khôi phục về trạng thái ban đầu). \\ \hline
\textbf{Audit log}             & Nhật ký hệ thống ghi lại mọi thao tác làm thay đổi dữ liệu nghiệp vụ kèm thông tin người thực hiện, thời điểm và dữ liệu trước, sau khi thay đổi. \\ \hline
\end{longtable}
